-- fibonacci with memoization
local memo = {}

local function fib(n)
  if n < 2 then return n end
  if memo[n] then return memo[n] end
  local value = fib(n - 1) + fib(n - 2)
  memo[n] = value
  return value
end

for i = 1, 10 do
  print(string.format("fib(%d) = %d", i, fib(i)))
end
